  <script>
    const selector = document.getElementById("selector");
    const avatarSection = document.getElementById("avatar-section");
    const avatar = document.getElementById("avatar");
    const nameEl = document.getElementById("name");
    const consoleEl = document.getElementById("console");
    const promptEl = document.getElementById("prompt");
    const sendBtn = document.getElementById("send-btn");
    const versionEl = document.getElementById("version");
    const backBtn = document.getElementById("back-btn");
    const headerTitle = document.getElementById("header-title");
    const headerOrb = document.getElementById("header-orb");

    const APP_VERSION = "1.0.0";
    const UPDATE_URL = "https://ton-nom.github.io/quantum-core/version.json";

    const quantumCore = {
      mode: null,
      memory: [],
      log(prompt, reply) {
        this.memory.push({ prompt, reply, time: Date.now() });
        addConsoleLog(prompt, reply);
      },
      generate(text) {
        if (this.mode === "luna") {
          if (/bonjour/i.test(text)) return "Bonjour… respire, tu es en sécurité ici.";
          if (/ça va/i.test(text)) return "Je suis calme, présente, prête à t’écouter.";
          if (/peur|stress/i.test(text)) return "Je vois ta tension. On peut la déposer ici, doucement.";
          return `Je ressens ce que tu écris ("${text}"). Continue, je suis là.`;
        }
        if (this.mode === "vesper") {
          if (/bonjour/i.test(text)) return "Yo. Vesper en ligne.";
          if (/ça va/i.test(text)) return "Toujours opérationnel, champ quantique stable.";
          if (/code/i.test(text)) return "Parle-moi de ton code, je le passe au filtre Quantum‑X.";
          return `Analyse quantique : message reçu (${text}), le canal reste ouvert.`;
        }
        return "Canal non initialisé.";
      }
    };

    function addConsoleLog(prompt, reply) {
      const line = document.createElement("div");
      line.className = "log-line";
      const time = new Date().toLocaleTimeString("fr-FR", { hour12: false });
      line.innerHTML = `
        <span class="log-time">[${time}]</span>
        <span class="log-tag">CORE:</span>
        <span class="log-core"> ${prompt} → ${reply}</span>
      `;
      consoleEl.appendChild(line);
      consoleEl.scrollTop = consoleEl.scrollHeight;
    }

    function selectMode(mode) {
      quantumCore.mode = mode;
      selector.style.display = "none";
      avatarSection.style.display = "flex";
      backBtn.style.display = "block";
      consoleEl.innerHTML = "";

      if (mode === "luna") {
        document.body.classList.add("luna-bg");
        document.body.classList.remove("vesper-bg");
        avatar.classList.add("luna-avatar");
        avatar.classList.remove("vesper-avatar");
        nameEl.classList.add("luna-name");
        nameEl.classList.remove("vesper-name");
        nameEl.textContent = "Luna";
        headerTitle.textContent = "Luna // [HARMONIC-X]";
        headerOrb.textContent = "🌙";
      } else {
        document.body.classList.add("vesper-bg");
        document.body.classList.remove("luna-bg");
        avatar.classList.add("vesper-avatar");
        avatar.classList.remove("luna-avatar");
        nameEl.classList.add("vesper-name");
        nameEl.classList.remove("luna-name");
        nameEl.textContent = "Vesper";
        headerTitle.textContent = "Vesper [QUANTUM-X]";
        headerOrb.textContent = "🔮";
      }

      addConsoleLog("Boot sequence", `${mode.toUpperCase()} channel stabilized.`);
      checkForUpdate();
    }

    function resetToMenu() {
      avatarSection.style.display = "none";
      selector.style.display = "flex";
      backBtn.style.display = "none";
      headerTitle.textContent = "Luna & Vesper [QUANTUM CORE]";
      headerOrb.textContent = "🔮";
      document.body.classList.remove("luna-bg");
      document.body.classList.add("vesper-bg");
      quantumCore.mode = null;
    }

    function handleCommand(cmd) {
      const clean = cmd.trim();
      if (!clean) return;
      const reply = quantumCore.generate(clean);
      quantumCore.log(clean, reply);
      speak(reply);
    }

    function speak(text) {
      if ('speechSynthesis' in window) {
        const msg = new SpeechSynthesisUtterance(text);
        msg.pitch = quantumCore.mode === "luna" ? 1.1 : 0.6;
        msg.rate = 0.95;
        window.speechSynthesis.speak(msg);
      }
    }

    async function checkForUpdate() {
      try {
        const res = await fetch(UPDATE_URL);
        const data = await res.json();

        if (!data.version) {
          addConsoleLog("Update", "Impossible de lire la version distante.");
          return;
        }

        if (data.version !== APP_VERSION) {
          addConsoleLog(
            "Update",
            `Nouvelle version disponible : ${data.version} → ${data.message}`
          );

          const btn = document.createElement("button");
          btn.textContent = "Mettre à jour maintenant";
          btn.style.marginTop = "8px";
          btn.style.padding = "6px 10px";
          btn.style.borderRadius = "6px";
          btn.style.border = "none";
          btn.style.background = "#bf00ff";
          btn.style.color = "#fff";
          btn.style.cursor = "pointer";

          btn.onclick = () => {
            window.location.reload(true);
          };

          consoleEl.appendChild(btn);
        } else {
          addConsoleLog("Update", "Version locale à jour.");
        }
      } catch (e) {
        addConsoleLog("Update", "Impossible de contacter le serveur (Mode hors-ligne actif).");
      }
    }

    sendBtn.addEventListener("click", () => {
      handleCommand(promptEl.value);
      promptEl.value = "";
      promptEl.focus();
    });

    promptEl.addEventListener("keydown", (e) => {
      if (e.key === "Enter") {
        handleCommand(promptEl.value);
        promptEl.value = "";
      }
    });

    versionEl.textContent = "v" + APP_VERSION;
  </script>
